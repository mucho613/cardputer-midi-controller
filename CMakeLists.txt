cmake_minimum_required(VERSION 3.16)
project(cardputer_midi_tests LANGUAGES CXX)
enable_testing()
add_executable(core_tests tests/core_tests.cpp lib/MidiCore/MidiCore.cpp lib/MidiCore/Preset.cpp)
target_include_directories(core_tests PRIVATE lib/MidiCore)
target_compile_features(core_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(core_tests PRIVATE /W4)
else()
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME core_tests COMMAND core_tests)
